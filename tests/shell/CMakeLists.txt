function(properdesktop_add_shell_test name)
    add_executable(${name} ${name}.cpp ${PROJECT_SOURCE_DIR}/tests/gtest_toolkit_main.cpp)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/tests)
    target_link_libraries(${name} PRIVATE GTest::gtest properdesktop-core properdesktop-test-loop
        ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
endfunction()
